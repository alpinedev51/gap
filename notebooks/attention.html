Load Forecasting with a Transformer

We build a sequence-to-sequence transformer that forecasts the next 24 hours of electricity demand using EIA 930 hourly grid data.

Pipeline overview:

  1. Pull data from the EIA Open Data API
  2. Preprocess & engineer features
  3. Build a torch.utils.data.Dataset
  4. Define the model: Time2Vec encoding → Transformer Encoder → Cross-Attention Decoder
  5. Train and evaluate
  6. Visualize attention weights

Step 3: Preprocessing & Feature Engineering

  • Use demand_adjusted_mwh as the forecast target
  • Add cyclical time features: hour, day-of-week, month as sin/cos pairs
  • lagged demand: lag_24 (same hour yesterday) and lag_168 (same hour last week)
  • Normalize with StandardScaler

BA = "CISO"   # balancing authority — change to MISO, PJM, ERCO, etc.

def preprocess(df_pudl, ba=BA):
    df = df_pudl[df_pudl["balancing_authority_code_eia"] == ba][
        ["datetime_utc", "demand_adjusted_mwh"]
    ].copy()
    df.columns = ["timestamp", "demand"]
    df["timestamp"] = pd.to_datetime(df["timestamp"], utc=True)
    df = df.sort_values("timestamp").set_index("timestamp")

    # cast to plain numpy float so interpolate(method="linear") works
    df["demand"] = df["demand"].astype("float64")

    # reindex to clean hourly grid and interpolate small gaps (≤ 3 h)
    full_idx = pd.date_range(df.index.min(), df.index.max(), freq="h", tz="UTC")
    df = df.reindex(full_idx)
    df["demand"] = df["demand"].interpolate(method="linear", limit=3)
    df = df.dropna(subset=["demand"])

    # lagged demand features
    df["lag_24"]  = df["demand"].shift(24)   # same hour yesterday
    df["lag_168"] = df["demand"].shift(168)  # same hour last week

    # cyclical time features
    df["hour_sin"]  = np.sin(2 * np.pi * df.index.hour / 24)
    df["hour_cos"]  = np.cos(2 * np.pi * df.index.hour / 24)
    df["dow_sin"]   = np.sin(2 * np.pi * df.index.dayofweek / 7)
    df["dow_cos"]   = np.cos(2 * np.pi * df.index.dayofweek / 7)
    df["month_sin"] = np.sin(2 * np.pi * df.index.month / 12)
    df["month_cos"] = np.cos(2 * np.pi * df.index.month / 12)

    df = df.dropna()   # drop the first 168 rows lost to lag_168
    return df

Step 4: PyTorch Dataset & DataLoader Each sample is a sliding window:

  • Encoder input: past LOOKBACK=168 hours of all features
  • Decoder input(tgt): the last 1 hour of demand (teacher-forcing seed)
  • Target (y): next HORIZON=24 hours of demand (normalized)
In [7]:
LOOKBACK  = 168   # encoder window: 1 week of hours
HORIZON   = 24    # decoder target: 24-hour forecast
BATCH     = 32

class LoadDataset(Dataset):
    def __init__(self, arr):
        # arr shape: (T, num_features)
        self.arr = torch.tensor(arr, dtype=torch.float32)

    def __len__(self):
        return len(self.arr) - LOOKBACK - HORIZON + 1

    def __getitem__(self, i):
        src = self.arr[i : i + LOOKBACK]                      # (LOOKBACK, F)
        # decoder seed: last observed demand step (shape 1, F)
        tgt_seed = self.arr[i + LOOKBACK - 1 : i + LOOKBACK]  # (1, F)
        # target: next HORIZON demand values only (index 0 = demand column)
        y   = self.arr[i + LOOKBACK : i + LOOKBACK + HORIZON, 0]  # (HORIZON,)
        return src, tgt_seed, y

Step 5: Model Architecture\n\n\nInput (B, T, F)\n │\n Linear projection → d_model\n │\n Time2Vec positional encoding\n │\n Transformer Encoder (self-attention over past 168 steps)\n │\n ┌──────────────────────┐\n │ Decoder query │ ← learned embeddings for each of the 24 future steps\n │ + Cross-Attention │ ← each future step attends to the full encoder output\n └──────────────────────┘\n │\n Linear → scalar demand forecast\n\n\n### Time2Vec\nInstead of fixed sinusoidal PE, Time2Vec learns both a linear (trend) component and $k$ periodic (sin) components:\n$$\\text{t2v}(t)_i = \\begin{cases} w_i t + b_i & i=0 \\\\ \\sin(w_i t + b_i) & i > 0 \\end{cases}$$

class Time2Vec(nn.Module): #module with 2 learnable vectors length 16, frequencies and phase shifts that we train
    """Learnable temporal embedding: 1 linear + (k-1) sinusoidal components."""
    def __init__(self, k: int):
        super().__init__()
        self.w = nn.Parameter(torch.randn(k))
        self.b = nn.Parameter(torch.randn(k))

    def forward(self, t): #takes integer positions and turns into 16-D vectors, 1 linear +15 periodic compoments
        # t: (B, T) — integer time indices
        t = t.unsqueeze(-1).float()          # (B, T, 1)
        linear = self.w[0] * t + self.b[0]  # to capture trend, 
        periodic = torch.sin(self.w[1:] * t + self.b[1:])  # (B, T, k-1)
        return torch.cat([linear, periodic], dim=-1)        # puts 1 liear and 15 periodic components into 1 vector

#defining our parameters like how many features, model dimension, number of heads, number of layers, feedforward dimension, dropout, horizon, and time2vec dimension
class LoadForecaster(nn.Module):
    def __init__(
        self,
        num_features: int = 7,   # F
        d_model:      int = 64,
        nhead:        int = 4,
        num_enc_layers: int = 2,
        num_dec_layers: int = 2,
        dim_ff:       int = 128,
        dropout:      float = 0.1,
        horizon:      int = 24,
        t2v_k:        int = 16,
    ):
        super().__init__()
        self.d_model = d_model
        self.horizon = horizon

        #  input projection: project raw features into 64-D model space, since Transformer operates in a vector space of dimension d_model
        self.input_proj = nn.Linear(num_features, d_model)

        #  Time2Vec , tells us what time the timestep is at, so the model can learn time-based patterns like daily or weekly seasonality
        #changeed from d2l logic since that one was language based and we need to encode time steps
        self.t2v     = Time2Vec(t2v_k)
        self.t2v_proj = nn.Linear(t2v_k, d_model)

        #  Transformer Encoder: multi-head self-attention + feedforward layers to encode the input sequence into a rich representation that captures temporal dependencies and patterns in the data
        # every hour attends every other hour, 4 heads means we have 4 different attention patterns looking at the data in parallel, 
        # and 2 layers means we stack this process twice to capture more complex relationships 
        enc_layer = nn.TransformerEncoderLayer(
            d_model=d_model, nhead=nhead,
            dim_feedforward=dim_ff, dropout=dropout, batch_first=True
        )
        self.encoder = nn.TransformerEncoder(enc_layer, num_layers=num_enc_layers)

        # Decoder: learned query embeddings for future steps
        #24 learnable 64-D vectors to learn representation of each hour for future 24 hours we want to predict, 
        # these queries will attend to the encoder output to generate the forecast
        self.query_embed = nn.Embedding(horizon, d_model)

        #transformer Decoder layer: each layer runs 2 attention operations
        #1. self attention among 24 query vectors
        #2. Cross attention: each future hour query attends to all 168 encoder outputs
        dec_layer = nn.TransformerDecoderLayer(
            d_model=d_model, nhead=nhead,
            dim_feedforward=dim_ff, dropout=dropout, batch_first=True
        )
        self.decoder = nn.TransformerDecoder(dec_layer, num_layers=num_dec_layers)

        # output head: final layer where we go back from 64-D model space to 1-D demand forecast for each future hour
        self.out_proj = nn.Linear(d_model, 1)

    def forward(self, src, return_attn=False):
        """
        src: (B, LOOKBACK, F)
        returns: (B, HORIZON)  — normalized demand forecast
        """
        B, T, _ = src.shape

        # project features to d_model
        x = self.input_proj(src)                             # (B, T, d_model)

        # add Time2Vec encoding
        t = torch.arange(T, device=src.device).unsqueeze(0).expand(B, -1)
        x = x + self.t2v_proj(self.t2v(t))                  # (B, T, d_model)

        # encode
        memory = self.encoder(x)                             # (B, T, d_model)

        # decoder queries: one learned embedding per future hour
        q_idx   = torch.arange(self.horizon, device=src.device)
        queries = self.query_embed(q_idx)                    # (HORIZON, d_model)
        queries = queries.unsqueeze(0).expand(B, -1, -1)     # (B, HORIZON, d_model)

        out = self.decoder(queries, memory)                  # (B, HORIZON, d_model)
        preds = self.out_proj(out).squeeze(-1)               # (B, HORIZON)

        return preds

EPOCHS = 10
LR     = 1e-3

optimizer = torch.optim.Adam(model.parameters(), lr=LR)
scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
    optimizer, patience=3, factor=0.5
)
criterion = nn.HuberLoss()

def run_epoch(dl, train=True):
    model.train(train)
    total_loss = 0.0
    with torch.set_grad_enabled(train):
        for src, _, y in dl:
            src, y = src.to(device), y.to(device)
            pred = model(src)
            loss = criterion(pred, y)
            if train:
                optimizer.zero_grad()
                loss.backward()
                nn.utils.clip_grad_norm_(model.parameters(), 1.0)
                optimizer.step()
            total_loss += loss.item() * len(src)
    return total_loss / len(dl.dataset)

train_losses, val_losses = [], []

for epoch in range(1, EPOCHS + 1):

def get_cross_attention(model, src_tensor):
    """
    Returns cross-attention weights from the last decoder layer.
    Shape: (HORIZON, LOOKBACK)  — averaged over heads.
    """
    model.eval()
    B, T, _ = src_tensor.shape
    x = model.input_proj(src_tensor)
    t = torch.arange(T, device=src_tensor.device).unsqueeze(0).expand(B, -1)
    x = x + model.t2v_proj(model.t2v(t))
    memory = model.encoder(x)

    q_idx   = torch.arange(model.horizon, device=src_tensor.device)
    queries = model.query_embed(q_idx).unsqueeze(0).expand(B, -1, -1)

    # manually run last decoder layer with need_weights=True
    last_layer = model.decoder.layers[-1]
    tgt = queries
    # run all but last decoder layer first
    for layer in model.decoder.layers[:-1]:
        tgt = layer(tgt, memory)
    # last layer: extract cross-attn weights
    tgt2, attn = last_layer.multihead_attn(
        tgt, memory, memory, need_weights=True, average_attn_weights=True
    )
    return attn.squeeze(0).detach().cpu().numpy()   # (HORIZON, LOOKBACK)
